SELECT fc.category,
       COUNT(r.rating)          AS rating_count,
       ROUND(AVG(r.rating), 2)  AS avg_rating
FROM fitness_class fc
INNER JOIN reservation r ON r.class_id = fc.id
WHERE r.rating IS NOT NULL

HAVING AVG(r.rating) >= 4.0
